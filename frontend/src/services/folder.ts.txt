// ===========================================
// Folder API Service
// ===========================================

import api from "./api";
import {
  Folder,
  CreateFolderRequest,
  UpdateFolderRequest,
} from "../types/folder";

/**
 * GET /folders
 * Fetch all folders for authenticated user
 */
export const getFolders = async (): Promise<Folder[]> => {
  const response = await api.get("/folders");

  // If backend returns array directly
  if (Array.isArray(response.data)) {
    return response.data;
  }

  // If backend wraps response in { data: [...] }
  return response.data.data;
};

/**
 * POST /folders
 * Create a new folder
 */
export const createFolder = async (
  folder: CreateFolderRequest
): Promise<Folder> => {
  const response = await api.post("/folders", folder);

  return response.data.data ?? response.data;
};

/**
 * PUT /folders/{id}
 * Rename folder
 */
export const renameFolder = async (
  id: number,
  folder: UpdateFolderRequest
): Promise<Folder> => {
  const response = await api.put(`/folders/${id}`, folder);

  return response.data.data ?? response.data;
};

/**
 * DELETE /folders/{id}
 * Delete folder
 */
export const deleteFolder = async (id: number): Promise<void> => {
  await api.delete(`/folders/${id}`);
};
